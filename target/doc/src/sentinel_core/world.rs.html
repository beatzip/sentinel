1use super::bomb::BombState;
2use super::evidence::Evidence;
3use super::feature::FeatureVector;
4use super::grenade::GrenadeState;
5use super::player::{PlayerId, PlayerState, Team};
6use super::round::RoundState;
7use super::tick::{Tick, TickState};
8
9/// Match context provides read-only access to all match data.
10/// This is passed to feature extractors and analyzers.
11pub struct MatchContext {
12    /// All tick states in order
13    states: Vec<TickState>,
14    /// Feature vectors computed for all players
15    feature_vectors: Vec<FeatureVector>,
16    /// Evidence collected during analysis
17    evidence: Vec<Evidence>,
18}
19
20impl MatchContext {
21    pub fn new(states: Vec<TickState>) -> Self {
22        Self {
23            states,
24            feature_vectors: Vec::new(),
25            evidence: Vec::new(),
26        }
27    }
28
29    /// Get the state at a specific tick
30    pub fn state_at(&self, tick: Tick) -> Option<&TickState> {
31        self.states.iter().find(|s| s.tick == tick)
32    }
33
34    /// Get all tick states
35    pub fn states(&self) -> &[TickState] {
36        &self.states
37    }
38
39    /// Get the first tick
40    pub fn first_tick(&self) -> Tick {
41        self.states.first().map(|s| s.tick).unwrap_or(Tick(0))
42    }
43
44    /// Get the last tick
45    pub fn last_tick(&self) -> Tick {
46        self.states.last().map(|s| s.tick).unwrap_or(Tick(0))
47    }
48
49    /// Get total number of ticks
50    pub fn tick_count(&self) -> usize {
51        self.states.len()
52    }
53
54    /// Get current round number (from the last state)
55    pub fn current_round(&self) -> u32 {
56        self.states
57            .last()
58            .map(|s| s.round.round_number)
59            .unwrap_or(0)
60    }
61
62    /// Get all alive players at a tick
63    pub fn alive_players_at(&self, tick: Tick) -> Vec<&PlayerState> {
64        self.state_at(tick)
65            .map(|s| s.players.iter().filter(|p| p.alive).collect())
66            .unwrap_or_default()
67    }
68
69    /// Get a specific player at a tick
70    pub fn player_at(&self, tick: Tick, player_id: PlayerId) -> Option<&PlayerState> {
71        self.state_at(tick)?
72            .players
73            .iter()
74            .find(|p| p.id == player_id)
75    }
76
77    /// Get all players on a team at a tick
78    pub fn team_players_at(&self, tick: Tick, team: Team) -> Vec<&PlayerState> {
79        self.state_at(tick)
80            .map(|s| {
81                s.players
82                    .iter()
83                    .filter(|p| p.team == team && p.alive)
84                    .collect()
85            })
86            .unwrap_or_default()
87    }
88
89    /// Get the bomb state at a tick
90    pub fn bomb_at(&self, tick: Tick) -> Option<&BombState> {
91        self.state_at(tick).map(|s| &s.bomb)
92    }
93
94    /// Get all active grenades at a tick
95    pub fn grenades_at(&self, tick: Tick) -> Vec<&GrenadeState> {
96        self.state_at(tick)
97            .map(|s| s.grenades.iter().filter(|g| g.active).collect())
98            .unwrap_or_default()
99    }
100
101    /// Get round state at a tick
102    pub fn round_at(&self, tick: Tick) -> Option<&RoundState> {
103        self.state_at(tick).map(|s| &s.round)
104    }
105
106    /// Add a feature vector
107    pub fn add_feature_vector(&mut self, fv: FeatureVector) {
108        self.feature_vectors.push(fv);
109    }
110
111    /// Get all feature vectors for a player
112    pub fn feature_vectors_for(&self, player: PlayerId) -> Vec<&FeatureVector> {
113        self.feature_vectors
114            .iter()
115            .filter(|fv| fv.player == player)
116            .collect()
117    }
118
119    /// Add evidence
120    pub fn add_evidence(&mut self, ev: Evidence) {
121        self.evidence.push(ev);
122    }
123
124    /// Get all evidence
125    pub fn evidence(&self) -> &[Evidence] {
126        &self.evidence
127    }
128
129    /// Get evidence for a specific player
130    pub fn evidence_for(&self, player: PlayerId) -> Vec<&Evidence> {
131        self.evidence
132            .iter()
133            .filter(|e| e.player == player)
134            .collect()
135    }
136}
