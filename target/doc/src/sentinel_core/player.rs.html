3/// Unique player identifier (SteamID)

50/// 3D position vector
51#[derive(Debug, Clone, Copy, Default, Serialize, Deserialize)]
52pub struct Vec3 {
53    pub x: f32,
54    pub y: f32,
55    pub z: f32,
56}
57
58impl Vec3 {
59    pub fn new(x: f32, y: f32, z: f32) -> Self {
60        Self { x, y, z }
61    }
62
63    pub fn distance_to(&self, other: &Vec3) -> f32 {
64        let dx = self.x - other.x;
65        let dy = self.y - other.y;
66        let dz = self.z - other.z;
67        (dx * dx + dy * dy + dz * dz).sqrt()
68    }
69
70    pub fn length(&self) -> f32 {
71        (self.x * self.x + self.y * self.y + self.z * self.z).sqrt()
72    }
73}
