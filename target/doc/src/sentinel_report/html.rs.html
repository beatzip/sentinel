6impl HtmlReport {
7    /// Generate an HTML report from match data
8    pub fn generate(report: &MatchReport) -> String {
9        let mut html = String::new();
10
11        // HTML header
12        html.push_str(r#"<!DOCTYPE html>
13<html lang="en">
14<head>
15    <meta charset="UTF-8">
16    <meta name="viewport" content="width=device-width, initial-scale=1.0">
17    <title>Sentinel AI Report</title>
18    <style>
19        * { margin: 0; padding: 0; box-sizing: border-box; }
20        body { font-family: -apple-system, BlinkMacSystemFont, 'Segoe UI', Roboto, sans-serif; background: #0f0f0f; color: #e0e0e0; padding: 20px; }
21        .container { max-width: 1200px; margin: 0 auto; }

23        h2 { color: #00ccff; margin: 30px 0 15px 0; border-bottom: 1px solid #333; padding-bottom: 10px; }
24        h3 { color: #ffaa00; margin: 20px 0 10px 0; }

38        th { color: #00ccff; }
39        .footer { margin-top: 40px; color: #666; font-size: 12px; }
40    </style>
41</head>
42<body>
43    <div class="container">
44"#);
45
46        // Title
47        html.push_str("        <h1>Sentinel AI Analysis Report</h1>\n");
48
49        // Metadata
50        html.push_str("        <div class=\"metadata\">\n");
51        html.push_str(&format!(
52            "            <p><strong>Map:</strong> {}</p>\n",
53            report.metadata.map_name
54        ));
55        html.push_str(&format!(
56            "            <p><strong>Server:</strong> {}</p>\n",
57            report.metadata.server_name
58        ));
59        html.push_str(&format!(
60            "            <p><strong>Rounds:</strong> {}</p>\n",

62        ));
63        html.push_str(&format!(
64            "            <p><strong>Duration:</strong> {:.1}s</p>\n",
65            report.metadata.duration_seconds

67        html.push_str(&format!(
68            "            <p><strong>Demo:</strong> {}</p>\n",

70        ));
71        html.push_str("        </div>\n\n");
72
73        // Overall score
74        html.push_str("        <h2>Overall Analysis</h2>\n");
75        let grade_class = Self::score_class(report.overall_anomaly);
76        let grade_text = Self::score_grade(report.overall_anomaly);
77        html.push_str(&format!(
78            "        <p>Anomaly Score: <span class=\"score-badge {}\">{:.2}/1.0 - {}</span></p>\n",
79            grade_class, report.overall_anomaly, grade_text
80        ));
81
82        // Player reports
83        html.push_str("        <h2>Player Analysis</h2>\n");

88            html.push_str("        <div class=\"player-card\">\n");

90                "            <h3>{} ({})</h3>\n",
91                player.name, player.steam_id
92            ));
93            html.push_str(&format!(
94                "            <p>Team: {} | Overall Score: <span class=\"score-badge {}\">{:.2}</span></p>\n",

98            // Category scores

101                "                <tr><th>Category</th><th>Score</th><th>Visual</th></tr>\n",
102            );
103            for (category, score) in &player.scores.categories {

105                html.push_str(&format!(

108                ));
109            }
110            html.push_str("            </table>\n");
111
112            // Evidence
113            if !player.evidence.is_empty() {
114                html.push_str(&format!(
115                    "            <p><strong>Evidence:</strong> {} items</p>\n",

118                html.push_str("            <table>\n");
119                html.push_str("                <tr><th>Tick</th><th>Feature</th><th>Score</th><th>Reason</th></tr>\n");
120
121                for ev in &player.evidence {
122                    html.push_str(&format!(
123                        "                <tr><td>{}</td><td>{}</td><td>{:.2}</td><td>{}</td></tr>\n",
124                        ev.tick.0, ev.feature, ev.score, ev.reason
125                    ));
126                }
127
128                html.push_str("            </table>\n");
129            }
130
131            html.push_str("        </div>\n\n");
132        }
133
134        // Footer
135        html.push_str(&format!(

137            report.version
138        ));
139
140        html.push_str("    </div>\n</body>\n</html>");
141
142        html
143    }

145    fn score_class(score: f64) -> &'static str {
146        if score < 0.2 {
147            "score-clean"
148        } else if score < 0.4 {
149            "score-low"
150        } else if score < 0.6 {
151            "score-moderate"
152        } else if score < 0.8 {
153            "score-high"
154        } else {
155            "score-critical"
156        }
157    }
158
159    fn score_grade(score: f64) -> &'static str {
160        if score < 0.2 {
161            "Clean"
162        } else if score < 0.4 {
163            "Low Suspicion"
164        } else if score < 0.6 {
165            "Moderate Suspicion"
166        } else if score < 0.8 {
167            "High Suspicion"
168        } else {
169            "Critical"
170        }
171    }

179    fn test_html_generation() {
180        use crate::{MatchMetadata, MatchReport, PlayerReport};
181        use sentinel_core::BehaviorScore;
182
183        let metadata = MatchMetadata {
184            demo_path: "test.dem".to_string(),
185            map_name: "de_dust2".to_string(),
186            server_name: "Test Server".to_string(),
187            total_rounds: 30,
188            duration_seconds: 1800.0,
189            tick_rate: 64,
190        };
191
192        let mut report = MatchReport::new(metadata);
193        report.add_player(PlayerReport {
194            steam_id: 12345,
195            name: "TestPlayer".to_string(),
196            team: "Terrorist".to_string(),
197            scores: BehaviorScore::new(),
198            evidence: Vec::new(),
199            summary: "No anomalies".to_string(),
200        });
201
202        let html = HtmlReport::generate(&report);
203        assert!(html.contains("de_dust2"));
204        assert!(html.contains("TestPlayer"));

206    }
