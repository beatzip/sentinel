7/// Calibration dataset containing feature distributions from known-legit matches

38impl CalibrationDataset {
39    pub fn new(name: impl Into<String>) -> Self {
40        Self {
41            name: name.into(),
42            version: "1.0.0".to_string(),
43            match_count: 0,
44            player_count: 0,
45            baselines: BaselineSet::new(),
46            metadata: DatasetMetadata {
47                created_at: chrono::Utc::now().to_rfc3339(),
48                source: "unknown".to_string(),
49                game_version: "cs2".to_string(),
50                tick_rate: 64,
51                maps: Vec::new(),
52            },
53        }
54    }

56    /// Create a default calibration dataset with CS2-typical values
57    pub fn default_cs2() -> Self {
58        let mut dataset = Self::new("cs2_default");
59        dataset.version = "1.0.0".to_string();
60        dataset.match_count = 1000;

66            "de_mirage".to_string(),
67            "de_inferno".to_string(),
68        ];
69        dataset
70    }

72    /// Save dataset to a JSON file

75        std::fs::write(path, json)

103/// Manager for golden test cases
104pub struct GoldenTestManager {
105    cases: Vec<GoldenTestCase>,
106    datasets_dir: PathBuf,
107}
108
109impl GoldenTestManager {
110    pub fn new(datasets_dir: PathBuf) -> Self {
111        Self {
112            cases: Vec::new(),
113            datasets_dir,
114        }
115    }
116
117    /// Load all golden test cases from the datasets directory
118    pub fn load_all(&mut self) -> Result<(), std::io::Error> {
119        let golden_dir = self.datasets_dir.join("golden");
120        if !golden_dir.exists() {
121            std::fs::create_dir_all(&golden_dir)?;
122            return Ok(());
123        }
124
125        for entry in std::fs::read_dir(golden_dir)? {
126            let entry = entry?;
127            let path = entry.path();
128            if path.extension().and_then(|s| s.to_str()) == Some("json") {
129                let json = std::fs::read_to_string(&path)?;
130                if let Ok(test_case) = serde_json::from_str::<GoldenTestCase>(&json) {
131                    self.cases.push(test_case);
132                }
133            }
134        }
135
136        Ok(())
137    }
138
139    /// Get all test cases
140    pub fn cases(&self) -> &[GoldenTestCase] {
141        &self.cases
142    }
143
144    /// Add a new test case
145    pub fn add_case(&mut self, case: GoldenTestCase) {
146        self.cases.push(case);
147    }
148
149    /// Save a test case to disk
150    pub fn save_case(&self, case: &GoldenTestCase) -> Result<(), std::io::Error> {
151        let golden_dir = self.datasets_dir.join("golden");
152        std::fs::create_dir_all(&golden_dir)?;
153
154        let path = golden_dir.join(format!("{}.json", case.name));
155        let json = serde_json::to_string_pretty(case).map_err(std::io::Error::other)?;
156        std::fs::write(path, json)
157    }
158
159    /// Run all golden tests and return results
160    pub fn run_all(&self) -> Vec<GoldenTestResult> {
161        self.cases
162            .iter()
163            .map(|case| {
164                // Check if demo file exists
165                let demo_exists = case.demo_path.exists();
166
167                GoldenTestResult {
168                    name: case.name.clone(),
169                    passed: demo_exists, // Simplified check
170                    message: if demo_exists {
171                        "Demo file exists".to_string()
172                    } else {
173                        format!("Demo file not found: {:?}", case.demo_path)
174                    },
175                }
176            })
177            .collect()
178    }
179}

189/// Dataset statistics for monitoring
190#[derive(Debug, Clone, Serialize, Deserialize)]
191pub struct DatasetStats {
192    /// Total number of feature vectors
193    pub total_vectors: usize,
194    /// Number of unique players
195    pub unique_players: usize,
196    /// Number of unique matches
197    pub unique_matches: usize,
198    /// Feature coverage (percentage of features with data)
199    pub feature_coverage: f64,
200    /// Per-feature statistics
201    pub feature_stats: BTreeMap<String, FeatureStats>,
202}
203
204#[derive(Debug, Clone, Serialize, Deserialize)]
205pub struct FeatureStats {
206    pub name: String,
207    pub count: usize,
208    pub mean: f64,
209    pub stddev: f64,
210    pub min: f64,
211    pub max: f64,
212}
213
214impl DatasetStats {
215    pub fn compute(vectors: &[sentinel_core::FeatureVector]) -> Self {
216        let mut feature_counts: BTreeMap<String, Vec<f64>> = BTreeMap::new();
217        let mut unique_players = std::collections::HashSet::new();
218        let mut unique_matches = std::collections::HashSet::new();
219
220        for fv in vectors {
221            unique_players.insert(fv.player);
222            unique_matches.insert(fv.round);
223
224            for (name, result) in &fv.features {
225                feature_counts
226                    .entry(name.clone())
227                    .or_default()
228                    .push(result.value);
229            }
230        }
231
232        let total_features = 20; // Expected number of features
233        let feature_coverage = feature_counts.len() as f64 / total_features as f64;
234
235        let feature_stats = feature_counts
236            .into_iter()
237            .map(|(name, values)| {
238                let count = values.len();
239                let mean = values.iter().sum::<f64>() / count as f64;
240                let variance =
241                    values.iter().map(|v| (v - mean).powi(2)).sum::<f64>() / count as f64;
242                let stddev = variance.sqrt();
243                let min = values.iter().cloned().fold(f64::INFINITY, f64::min);
244                let max = values.iter().cloned().fold(f64::NEG_INFINITY, f64::max);
245
246                (
247                    name.clone(),
248                    FeatureStats {
249                        name,
250                        count,
251                        mean,
252                        stddev,
253                        min,
254                        max,
255                    },
256                )
257            })
258            .collect();
259
260        Self {
261            total_vectors: vectors.len(),
262            unique_players: unique_players.len(),
263            unique_matches: unique_matches.len(),
264            feature_coverage,
265            feature_stats,
266        }
267    }
268}
