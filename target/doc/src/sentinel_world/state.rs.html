6/// Complete world state at a specific point in time
7#[derive(Debug, Clone)]
8pub struct WorldState {
9    /// Current tick
10    pub tick: Tick,
11    /// All players
12    pub players: BTreeMap<PlayerId, PlayerState>,
13    /// All active grenades
14    pub grenades: Vec<GrenadeState>,
15    /// Bomb state
16    pub bomb: BombState,
17    /// Current round
18    pub round: RoundState,
19    /// Kill feed for current round
20    pub kill_feed: Vec<KillEvent>,
21}

23/// Kill event for tracking
24#[derive(Debug, Clone)]
25pub struct KillEvent {
26    pub tick: Tick,
27    pub attacker: PlayerId,
28    pub victim: PlayerId,
29    pub weapon: String,
30    pub headshot: bool,
31    pub assisted: bool,
32    pub assist_player: Option<PlayerId>,
33}

35impl WorldState {
36    /// Create a new empty world state
37    pub fn new(tick: Tick) -> Self {
38        Self {
39            tick,
40            players: BTreeMap::new(),
41            grenades: Vec::new(),
42            bomb: BombState::Carried {
43                carrier: PlayerId::new(0),
44            },
45            round: RoundState {
46                round_number: 1,
47                phase: RoundPhase::Freezetime,
48                clock: 115.0,
49                t_score: 0,
50                ct_score: 0,
51                winner: None,
52            },
53            kill_feed: Vec::new(),
54        }
55    }
56
57    /// Get a player by ID
58    pub fn player(&self, id: PlayerId) -> Option<&PlayerState> {
59        self.players.get(&id)
60    }
61
62    /// Get all alive players
63    pub fn alive_players(&self) -> Vec<&PlayerState> {
64        self.players.values().filter(|p| p.alive).collect()
65    }
66
67    /// Get players on a specific team
68    pub fn team_players(&self, team: sentinel_core::Team) -> Vec<&PlayerState> {
69        self.players
70            .values()
71            .filter(|p| p.team == team && p.alive)
72            .collect()
73    }
74
75    /// Get the number of alive players on each team
76    pub fn alive_counts(&self) -> (usize, usize) {
77        let t_count = self
78            .players
79            .values()
80            .filter(|p| p.team == sentinel_core::Team::Terrorist && p.alive)
81            .count();
82        let ct_count = self
83            .players
84            .values()
85            .filter(|p| p.team == sentinel_core::Team::CounterTerrorist && p.alive)
86            .count();
87        (t_count, ct_count)
88    }
89
90    /// Get all active grenades
91    pub fn active_grenades(&self) -> Vec<&GrenadeState> {
92        self.grenades.iter().filter(|g| g.active).collect()
93    }
94
95    /// Get grenades of a specific type
96    pub fn grenades_by_type(&self, grenade_type: GrenadeType) -> Vec<&GrenadeState> {
97        self.grenades
98            .iter()
99            .filter(|g| g.active && g.grenade_type == grenade_type)
100            .collect()
101    }
102
103    /// Check if a player is visible from another player's position
104    /// This is a simplified check - real implementation would use raycasting
105    pub fn is_visible(&self, observer: PlayerId, target: PlayerId) -> bool {
106        let observer = match self.player(observer) {
107            Some(p) => p,
108            None => return false,
109        };
110        let target = match self.player(target) {
111            Some(p) => p,
112            None => return false,
113        };
114
115        if !target.alive {
116            return false;
117        }
118
119        // Simple distance-based visibility check
120        let distance = observer.position.distance_to(&target.position);
121        distance < 1000.0 // Simplified threshold
122    }
123
124    /// Get the closest enemy to a player
125    pub fn closest_enemy(&self, player_id: PlayerId) -> Option<&PlayerState> {
126        let player = self.player(player_id)?;
127        let target_team = match player.team {
128            sentinel_core::Team::Terrorist => sentinel_core::Team::CounterTerrorist,
129            sentinel_core::Team::CounterTerrorist => sentinel_core::Team::Terrorist,
130            sentinel_core::Team::Unassigned => return None,
131        };
132
133        self.team_players(target_team).into_iter().min_by(|a, b| {
134            let dist_a = player.position.distance_to(&a.position);
135            let dist_b = player.position.distance_to(&b.position);
136            dist_a
137                .partial_cmp(&dist_b)
138                .unwrap_or(std::cmp::Ordering::Equal)
139        })
140    }
141
142    /// Add a kill event to the kill feed
143    pub fn add_kill(&mut self, kill: KillEvent) {
144        self.kill_feed.push(kill);
145    }
146
147    /// Get all kills in the current round
148    pub fn round_kills(&self) -> Vec<&KillEvent> {
149        self.kill_feed.iter().collect()
150    }
151}
