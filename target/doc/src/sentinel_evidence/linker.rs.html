1use std::collections::BTreeMap;
2
3use sentinel_core::{Evidence, MatchContext};
4
5/// Links evidence to contextual information for explainability
6pub struct EvidenceLinker;
7
8impl EvidenceLinker {
9    /// Enrich evidence with additional context from the match
10    pub fn link(evidence: &mut Evidence, ctx: &MatchContext) {
11        // Add visibility context
12        if let Some(state) = ctx.state_at(evidence.tick) {
13            // Check if the player was visible at this tick
14            if let Some(player) = state.players.iter().find(|p| p.id == evidence.player) {
15                evidence
16                    .metadata
17                    .insert("player_alive".to_string(), player.alive.to_string());
18                evidence
19                    .metadata
20                    .insert("player_health".to_string(), player.health.to_string());
21                evidence
22                    .metadata
23                    .insert("player_weapon".to_string(), format!("{:?}", player.weapon));
24            }
25
26            // Add round context
27            evidence.metadata.insert(
28                "round_number".to_string(),
29                state.round.round_number.to_string(),
30            );
31            evidence.metadata.insert(
32                "round_clock".to_string(),
33                format!("{:.1}", state.round.clock),
34            );
35            evidence
36                .metadata
37                .insert("t_score".to_string(), state.round.t_score.to_string());
38            evidence
39                .metadata
40                .insert("ct_score".to_string(), state.round.ct_score.to_string());
41        }
42
43        // Add tick timestamp in seconds
44        evidence.metadata.insert(
45            "tick_seconds".to_string(),
46            format!("{:.2}", evidence.tick.as_seconds()),
47        );
48    }
49
50    /// Link all evidence in a collection
51    pub fn link_all(evidence: &mut [Evidence], ctx: &MatchContext) {
52        for ev in evidence.iter_mut() {
53            Self::link(ev, ctx);
54        }
55    }
56
57    /// Generate a human-readable explanation for evidence
58    pub fn explain(evidence: &Evidence) -> String {
59        let tick_seconds = evidence
60            .metadata
61            .get("tick_seconds")
62            .map(|s| s.as_str())
63            .unwrap_or("?");
64
65        let round = evidence
66            .metadata
67            .get("round_number")
68            .map(|s| s.as_str())
69            .unwrap_or("?");
70
71        let weapon = evidence
72            .metadata
73            .get("player_weapon")
74            .map(|s| s.as_str())
75            .unwrap_or("unknown");
76
77        format!(
78            "Round {}, tick {} ({}s): {} score {:.2} - {} [weapon: {}]",
79            round,
80            evidence.tick.0,
81            tick_seconds,
82            evidence.feature,
83            evidence.score,
84            evidence.reason,
85            weapon
86        )
87    }
88
89    /// Generate a summary explanation for a player's evidence
90    pub fn summarize_player_evidence(evidence: &[&Evidence]) -> String {
91        if evidence.is_empty() {
92            return "No anomalous behavior detected.".to_string();
93        }
94
95        // Group by feature
96        let mut by_feature: BTreeMap<&str, Vec<&Evidence>> = BTreeMap::new();
97        for ev in evidence {
98            by_feature.entry(ev.feature.as_str()).or_default().push(ev);
99        }
100
101        let mut summary = format!("Found {} evidence items:\n", evidence.len());
102
103        for (feature, items) in &by_feature {
104            let avg_score = items.iter().map(|e| e.score).sum::<f64>() / items.len() as f64;
105            summary.push_str(&format!(
106                "  - {}: {} occurrences, avg score {:.2}\n",
107                feature,
108                items.len(),
109                avg_score
110            ));
111        }
112
113        summary
114    }
115}
